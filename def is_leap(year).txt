def is_leap(year):
    leap = False
    
    # Check if the year is divisible by 4
    if year % 4 == 0:
        # If it's a century  year, it must also be divisible by 400
        if year % 100 == 0:
            if year % 400 == 0:
                leap = True
            else:
                leap = False
        else:
            # Divisible by 4 but not by 100 -> It's a leap year
            leap = True
            
    return leap
year = int(input())
print(is_leap(year))
